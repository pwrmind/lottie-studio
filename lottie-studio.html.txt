/* ============================================================================
 * 2. ПАРСЕР SQL-DSL
 * ============================================================================ */
function splitTopLevel(str, sep = ',') {
  const res = [];
  let cur = '', quote = null, depth = 0;
  for (let i = 0; i < str.length; i++) {
    const ch = str[i];
    if (quote) {
      cur += ch;
      if (ch === quote && str[i - 1] !== '\\') quote = null;
    } else {
//Use code with caution.
if (ch === "'" || ch === '"') quote = ch;
else if (ch === '(') depth++;
else if (ch === ')') depth--;
if (ch === sep && depth === 0) {
res.push(cur.trim());
cur = '';
continue;
}
cur += ch;
}
}
if (cur.trim()) res.push(cur.trim());
return res;
}
function parseValue(raw) {
const s = raw.trim();
if ((s.startsWith("'") && s.endsWith("'")) || (s.startsWith('"') && s.endsWith('"'))) {
return s.slice(1, -1);
}
if (s === 'true') return true;
if (s === 'false') return false;
if (!isNaN(Number(s)) && s !== '') return Number(s);
return s;
}
function parseAssignments(str) {
const obj = {};
for (const part of splitTopLevel(str, ',')) {
const eq = part.indexOf('=');
if (eq === -1) throw new Error(`Ожидалось присваивание "ключ = значение" в: ${part}`);
const k = part.slice(0, eq).trim();
const v = part.slice(eq + 1).trim();
obj[k] = parseValue(v);
}
return obj;
}
function parseDSL(sqlText) {
// Удаляем однострочные комментарии --
const cleaned = sqlText
.split('\n')
.map(line => {
let inStr = false, q = '';
for (let i = 0; i < line.length - 1; i++) {
const c = line[i];
if (inStr) { if (c === q) inStr = false; }
else if (c === "'" || c === '"') { inStr = true; q = c; }
else if (c === '-' && line[i + 1] === '-') return line.slice(0, i);
}
return line;
})
.join('\n');
const statements = splitTopLevel(cleaned, ';');
const comp = { width: 540, height: 960, fps: 60, duration: 5.0 };
const layers = new Map();
const updates = [];
for (let idx = 0; idx < statements.length; idx++) {
const stmt = statements[idx].trim();
if (!stmt) continue;
// 1. INSERT INTO layers (cols) VALUES (vals), (vals)...
const insMatch = stmt.match(/^INSERT\s+INTO\s+layers\s*\(([^)]+)\)\sVALUES\s(.+)$/is);
if (insMatch) {
const cols = splitTopLevel(insMatch[1], ',').map(c => c.trim());
const rowsRaw = insMatch.trim();
// Разбираем кортежи (...)
const tuples = [];
let depth = 0, start = -1, quote = null;
for (let i = 0; i < rowsRaw.length; i++) {
const ch = rowsRaw[i];
if (quote) { if (ch === quote) quote = null; continue; }
if (ch === "'" || ch === '"') { quote = ch; continue; }
if (ch === '(') { if (depth === 0) start = i + 1; depth++; }
else if (ch === ')') {
depth--;
if (depth === 0 && start !== -1) {
tuples.push(rowsRaw.slice(start, i));
start = -1;
}
}
}
for (const tup of tuples) {
const vals = splitTopLevel(tup, ',').map(parseValue);
if (vals.length !== cols.length) {
throw new Error(`Количество значений (${vals.length}) не совпадает с колонками (${cols.length})`);
}
const rowObj = {
type: 'rect', x: 0, y: 0, width: 100, height: 100,
scale: 100, rotation: 0, opacity: 100, color: '#ffffff',
radius: 0, text: '', size: 24
};
cols.forEach((c, i) => { rowObj[c] = vals[i]; });
if (!rowObj.id) throw new Error('У каждого слоя в INSERT должен быть указан id');
layers.set(rowObj.id, rowObj);
}
continue;
}
// 2. UPDATE composition SET ...
const compMatch = stmt.match(/^UPDATE\s+composition\s+SET\s+(.+)$/is);
if (compMatch) {
Object.assign(comp, parseAssignments(compMatch[1]));
continue;
}
// 3. UPDATE layers SET ... [WHERE ...] [AT t] [WITH (...)]
const updMatch = stmt.match(
/^UPDATE\s+layers\s+SET\s+(.+?)(?:\s+WHERE\s+(.+?))?(?:\s+AT\s+([\d.]+))?(?:\s+WITH\s*\((.+)\))?$/is
);
if (updMatch) {
const setObj = parseAssignments(updMatch[1]);
const whereRaw = updMatch;
const atTime = updMatch !== undefined ? parseFloat(updMatch) : 0;
const withObj = updMatch[4] ? parseAssignments(updMatch[4]) : {};
let targetIds = null; // null = все слои
if (whereRaw) {
const inMatch = whereRaw.match(/^id\s+IN\s*\((.+)\)$/i);
const eqMatch = whereRaw.match(/^id\s*=\s*(.+)$/i);
if (inMatch) {
targetIds = splitTopLevel(inMatch[1], ',').map(parseValue);
} else if (eqMatch) {
targetIds = [parseValue(eqMatch[1])];
} else {
throw new Error(`Неподдерживаемое условие WHERE: ${whereRaw}`);
}
}
updates.push({
t: atTime,
dur: Number(withObj.duration ?? withObj.dur ?? 0),
ease: String(withObj.ease ?? 'inOut'),
targetIds,
set: setObj
});
continue;
}
throw new Error(`Неизвестная инструкция SQL: ${stmt.slice(0, 60)}...`);
}
return { comp, layers, updates };
}

const EASING_MAP = {
linear: { i: { x:, y: [1] },       o: { x:, y: [0] } },
in:     { i: { x:, y: [1] },       o: { x: [0.42], y: [0] } },
out:    { i: { x: [0.58], y: [1] },    o: { x:, y: [0] } },
inOut:  { i: { x: [0.42], y: [1] },    o: { x: [0.58], y: [0] } }
};
function hexToLottieColor(hex) {
let c = String(hex).replace('#', '').trim();
if (c.length === 3) c = c.split('').map(x => x + x).join('');
const num = parseInt(c, 16);
if (isNaN(num)) return;
return [
+(((num >> 16) & 255) / 255).toFixed(4),
+(((num >> 8) & 255) / 255).toFixed(4),
+((num & 255) / 255).toFixed(4),
1
];
}
function buildKeyframes(events, initialVal, fps, isArray = false) {
// Сортируем события по времени старта
const sorted = [...events].sort((a, b) => a.t - b.t);
if (sorted.length === 0) {
return { a: 0, k: isArray ? initialVal : initialVal };
}
const kf = [];
let curVal = initialVal;
for (const ev of sorted) {
const startFrame = Math.round(ev.t * fps);
const durFrames = Math.round(ev.dur * fps);
const endFrame = startFrame + durFrames;
const ease = EASING_MAP[ev.ease] || EASING_MAP.inOut;
if (durFrames <= 0) {
// Мгновенное изменение (Hold keyframe)
kf.push({
t: startFrame,
s: isArray ? [...ev.val] : [ev.val],
h: 1
});
curVal = ev.val;
} else {
// Плавная интерполяция: ставим начальный ключ (если ещё не стоит на этом кадре) и конечный
const last = kf[kf.length - 1];
if (!last || last.t < startFrame) {
kf.push({
t: startFrame,
s: isArray ? [...curVal] : [curVal],
i: ease.i,
o: ease.o
});
} else if (last.t === startFrame) {
delete last.h;
last.i = ease.i;
last.o = ease.o;
}
kf.push({
t: endFrame,
s: isArray ? [...ev.val] : [ev.val]
});
curVal = ev.val;
}
}
return { a: 1, k: kf };
}
function compileToLottie({ comp, layers, updates }) {
const fps = comp.fps || 60;
const totalFrames = Math.max(1, Math.round((comp.duration || 5) * fps));
const lottieLayers = [];
let ind = 1;
// Lottie рендерит слои сверху вниз (индекс 0 — самый верхний слой),
// поэтому разворачиваем порядок из INSERT, чтобы первые слои были фоном.
const layerList = Array.from(layers.values()).reverse();
for (const layer of layerList) {
const myUpdates = updates.filter(u => !u.targetIds || u.targetIds.includes(layer.id));
// Собираем события по отдельным свойствам
const posEvents = [];
const scaleEvents = [];
const rotEvents = [];
const opEvents = [];
const colorEvents = [];
let curX = layer.x, curY = layer.y;
let curScale = layer.scale;
for (const u of myUpdates) {
if (u.set.x !== undefined || u.set.y !== undefined) {
const nx = u.set.x !== undefined ? u.set.x : curX;
const ny = u.set.y !== undefined ? u.set.y : curY;
posEvents.push({ t: u.t, dur: u.dur, ease: u.ease, val: [nx, ny, 0] });
curX = nx; curY = ny;
}
if (u.set.scale !== undefined) {
curScale = u.set.scale;
scaleEvents.push({ t: u.t, dur: u.dur, ease: u.ease, val: [curScale, curScale, 100] });
}
if (u.set.rotation !== undefined) {
rotEvents.push({ t: u.t, dur: u.dur, ease: u.ease, val: u.set.rotation });
}
if (u.set.opacity !== undefined) {
opEvents.push({ t: u.t, dur: u.dur, ease: u.ease, val: u.set.opacity });
}
if (u.set.color !== undefined) {
colorEvents.push({ t: u.t, dur: u.dur, ease: u.ease, val: hexToLottieColor(u.set.color) });
}
}
const ks = {
o: buildKeyframes(opEvents, layer.opacity, fps, false),
r: buildKeyframes(rotEvents, layer.rotation, fps, false),
p: buildKeyframes(posEvents, [layer.x, layer.y, 0], fps, true),
a: { a: 0, k: [0, 0, 0] },
s: buildKeyframes(scaleEvents, [layer.scale, layer.scale, 100], fps, true)
};
const baseLayer = {
ddd: 0,
ind: ind++,
nm: layer.id,
sr: 1,
ks,
ao: 0,
ip: 0,
op: totalFrames,
st: 0,
bm: 0
};
if (layer.type === 'text') {
lottieLayers.push({
...baseLayer,
ty: 5, // Text layer
t: {
d: {
k: [{
s: {
s: layer.size || 24,
f: 'Arial',
t: String(layer.text || ''),
j: 2, // Center align
tr: 0,
lh: (layer.size || 24) * 1.2,
ls: 0,
fc: hexToLottieColor(layer.color).slice(0, 3)
},
t: 0
}]
},
p: {}, m: { g: 1, a: { a: 0, k: [0, 0] } }, a: []
}
});
} else {
// Shape layer (rect или ellipse)
const shapeGeom = layer.type === 'ellipse'
? {
ty: 'el',
d: 1,
s: { a: 0, k: [layer.width, layer.height] },
p: { a: 0, k: [0, 0] },
nm: 'Ellipse'
}
: {
ty: 'rc',
d: 1,
s: { a: 0, k: [layer.width, layer.height] },
p: { a: 0, k: [0, 0] },
r: { a: 0, k: layer.radius || 0 },
nm: 'Rectangle'
};
const fillProp = {
ty: 'fl',
c: buildKeyframes(colorEvents, hexToLottieColor(layer.color), fps, true),
o: { a: 0, k: 100 },
r: 1,
bm: 0,
nm: 'Fill'
};
lottieLayers.push({
...baseLayer,
ty: 4, // Shape layer
shapes: [{
ty: 'gr',
it: [
shapeGeom,
fillProp,
{
ty: 'tr',
p: { a: 0, k: [0, 0] },
a: { a: 0, k: [0, 0] },
s: { a: 0, k: [100, 100] },
r: { a: 0, k: 0 },
o: { a: 0, k: 100 }
}
],
nm: 'Group'
}]
});
}
}
return {
v: '5.12.2',
fr: fps,
ip: 0,
op: totalFrames,
w: comp.width,
h: comp.height,
nm: 'SQL_DSL_Animation',
ddd: 0,
assets: [],
layers: lottieLayers
};
}

const STORAGE_KEY = 'sql_lottie_studio:dsl';
const editor = document.getElementById('dsl-editor');
const errorBox = document.getElementById('error-box');
const lottieBox = document.getElementById('lottie-box');
const stageWrap = document.getElementById('stage-wrap');
const rawJsonView = document.getElementById('raw-json-view');
const scrubber = document.getElementById('scrubber');
const timecode = document.getElementById('timecode');
const btnPlay = document.getElementById('btn-play');
const recBadge = document.getElementById('rec-badge');
let anim = null;
let currentLottieData = null;
let isPlaying = true;
let isExporting = false;
editor.value = localStorage.getItem(STORAGE_KEY) || DEFAULT_DSL;
function fitStageBox(w, h) {
const wrapRect = stageWrap.getBoundingClientRect();
const maxW = Math.max(120, wrapRect.width - 40);
const maxH = Math.max(120, wrapRect.height - 40);
const scale = Math.min(maxW / w, maxH / h, 1);
lottieBox.style.width = Math.round(w * scale) + 'px';
lottieBox.style.height = Math.round(h * scale) + 'px';
}
window.addEventListener('resize', () => {
if (currentLottieData) fitStageBox(currentLottieData.w, currentLottieData.h);
});
function updateTimeUI() {
if (!anim || !currentLottieData) return;
const frame = anim.currentFrame;
const total = anim.totalFrames;
const fps = currentLottieData.fr;
scrubber.value = Math.round(frame);
timecode.textContent = ${(frame / fps).toFixed(2)} / ${(total / fps).toFixed(2)}s;
}
function compileAndRun() {
try {
const parsed = parseDSL(editor.value);
currentLottieData = compileToLottie(parsed);
localStorage.setItem(STORAGE_KEY, editor.value);
errorBox.classList.remove('visible');
errorBox.textContent = '';
// Обновляем Raw JSON панель
rawJsonView.textContent = JSON.stringify(currentLottieData, null, 2);
// Перезапускаем плеер lottie-web
if (anim) anim.destroy();
lottieBox.innerHTML = '';
fitStageBox(currentLottieData.w, currentLottieData.h);
anim = lottie.loadAnimation({
container: lottieBox,
renderer: 'svg',
loop: true,
autoplay: true,
animationData: JSON.parse(JSON.stringify(currentLottieData))
});
isPlaying = true;
btnPlay.textContent = '⏸ Пауза';
scrubber.max = Math.max(1, currentLottieData.op - 1);
anim.addEventListener('enterFrame', updateTimeUI);
} catch (err) {
errorBox.textContent = 'Ошибка компиляции: ' + err.message;
errorBox.classList.add('visible');
}
}
// Переключение вкладок Превью / Raw Lottie
const tabPreview = document.getElementById('tab-preview');
const tabRaw = document.getElementById('tab-raw');
tabPreview.addEventListener('click', () => {
tabPreview.classList.add('active');
tabRaw.classList.remove('active');
stageWrap.style.display = 'flex';
rawJsonView.style.display = 'none';
});
tabRaw.addEventListener('click', () => {
tabRaw.classList.add('active');
tabPreview.classList.remove('active');
stageWrap.style.display = 'none';
rawJsonView.style.display = 'block';
});
// Управление воспроизведением
btnPlay.addEventListener('click', () => {
if (!anim || isExporting) return;
if (isPlaying) {
anim.pause();
isPlaying = false;
btnPlay.textContent = '▶ Старт';
} else {
anim.play();
isPlaying = true;
btnPlay.textContent = '⏸ Пауза';
}
});
scrubber.addEventListener('input', () => {
if (!anim || isExporting) return;
anim.pause();
isPlaying = false;
btnPlay.textContent = '▶ Старт';
anim.goToAndStop(Number(scrubber.value), true);
updateTimeUI();
});
// Сохранение и загрузка DSL / Lottie
function downloadBlob(blob, filename) {
const a = document.createElement('a');
a.href = URL.createObjectURL(blob);
a.download = filename;
document.body.appendChild(a);
a.click();
setTimeout(() => { URL.revokeObjectURL(a.href); a.remove(); }, 2000);
}
document.getElementById('btn-compile').addEventListener('click', compileAndRun);
document.getElementById('btn-save-dsl').addEventListener('click', () => {
downloadBlob(new Blob([editor.value], { type: 'text/plain;charset=utf-8' }), 'animation.sql');
});
document.getElementById('btn-load-dsl').addEventListener('click', () => {
document.getElementById('file-dsl').click();
});
document.getElementById('file-dsl').addEventListener('change', e => {
const file = e.target.files[0];
if (!file) return;
const reader = new FileReader();
reader.onload = ev => {
editor.value = ev.target.result;
compileAndRun();
};
reader.readAsText(file);
e.target.value = '';
});
document.getElementById('btn-reset-dsl').addEventListener('click', () => {
editor.value = DEFAULT_DSL;
compileAndRun();
});
document.getElementById('btn-save-lottie').addEventListener('click', () => {
if (!currentLottieData) return;
const json = JSON.stringify(currentLottieData, null, 2);
downloadBlob(new Blob([json], { type: 'application/json' }), 'animation.lottie.json');
});
editor.addEventListener('keydown', e => {
if (e.code === 'Enter' && (e.ctrlKey || e.metaKey)) {
e.preventDefault();
compileAndRun();
}
});

document.getElementById('btn-export-video').addEventListener('click', async () => {
if (!currentLottieData || isExporting) return;
const mimeCandidates = [
'video/webm;codecs=vp9',
'video/webm;codecs=vp8',
'video/webm',
'video/mp4'
];
const mime = mimeCandidates.find(m => window.MediaRecorder && MediaRecorder.isTypeSupported(m));
if (!mime) {
alert('Ваш браузер не поддерживает MediaRecorder для записи видео.');
return;
}
isExporting = true;
if (anim) anim.pause();
isPlaying = false;
btnPlay.textContent = '▶ Старт';
recBadge.classList.add('on');
document.getElementById('btn-export-video').disabled = true;
const w = currentLottieData.w;
const h = currentLottieData.h;
const fps = currentLottieData.fr;
const totalFrames = currentLottieData.op;
// Создаём скрытый контейнер и рендерим через canvas-рендерер lottie-web
const offDiv = document.createElement('div');
offDiv.style.cssText = position:fixed;left:-9999px;top:0;width:${w}px;height:${h}px;;
document.body.appendChild(offDiv);
const offAnim = lottie.loadAnimation({
container: offDiv,
renderer: 'canvas',
loop: false,
autoplay: false,
animationData: JSON.parse(JSON.stringify(currentLottieData)),
rendererSettings: { clearCanvas: true, preserveDrawingBuffer: true }
});
await new Promise(r => offAnim.addEventListener('DOMLoaded', r, { once: true }));
const canvas = offDiv.querySelector('canvas');
const stream = canvas.captureStream(0);
const track = stream.getVideoTracks()[0];
const recorder = new MediaRecorder(stream, { mimeType: mime, videoBitsPerSecond: 8_000_000 });
const chunks = [];
recorder.ondataavailable = e => { if (e.data.size > 0) chunks.push(e.data); };
const donePromise = new Promise(resolve => { recorder.onstop = resolve; });
recorder.start();
const frameDelay = 1000 / fps;
for (let f = 0; f < totalFrames; f++) {
offAnim.goToAndStop(f, true);
if (track.requestFrame) track.requestFrame();
scrubber.value = f;
timecode.textContent = ${(f / fps).toFixed(2)} / ${(totalFrames / fps).toFixed(2)}s;
await new Promise(r => setTimeout(r, frameDelay));
}
recorder.stop();
await donePromise;
offAnim.destroy();
offDiv.remove();
const ext = mime.includes('mp4') ? 'mp4' : 'webm';
downloadBlob(new Blob(chunks, { type: mime }), lottie_export.${ext});
isExporting = false;
recBadge.classList.remove('on');
document.getElementById('btn-export-video').disabled = false;
if (anim) { anim.play(); isPlaying = true; btnPlay.textContent = '⏸ Пауза'; }
});
// Старт при загрузке
compileAndRun();

</script>
</body>
